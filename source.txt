FOR pass%=0 TO 2 STEP 2
  P%=spiSendCode% : [ OPT pass%
  .spiSend
  ; R0  = Command
  ; R1 -> Start of data
  ; R5 -> End of data
  SWI   "OS_EnterOS"
  :
  LDR   R3,gpiov                   ;
  MOV   R4,#1                      ;
  MOV   R2,R4,LSL #25              ;
  STR   R2,[R3,#40]                ; Set D/C Low
  MCR   cp15,0,R3,c7,c10,5         ; Data Memory Barrier
  :
  LDR   R3,spiv
  LDR   R2,[R3,#0]             ; Read current SPI CS (Control/Status)
  AND   R2,R2,#&0c             ; Preserve only bits 2-3:
                               ; CPHA (clock phase),  CPOL (clock polarity)
  ORR   R2,R2,#&6b0            ; Set b4=CLEAR_TX
                               ;     b5=CLEAR_RX (flush both FIFOs)
                               ;     b7=TA (Transfer Active - starts xfr)
                               ;     b8=DMAEN (DMA Enable)
                               ;     b9=INTD (interrupt on done)
                               ;     b0=INTR (int when RX FIFO needs reading)
  STR   R2,[R3,#0]             ; Write updated control word back to SPI CS reg
  :
  MCR   cp15,0,R3,c7,c10,5
  STR   R0,[R3,#4]
  :
  .cmdwait
  MCR   cp15,0,R3,c7,c10,5
  LDR   R2,[R3,#0]
  TST   R2,#1<<16
  BEQ   cmdwait
  TST   R2,#1<<17
  LDRNE R0,[R3,#4]
  ;ORR   R2,R2,#&30
  ;STR   R2,[R3,#0]
  :
  LDR   R3,gpiov                   ;
  MOV   R4,#1                      ;
  MOV   R2,R4,LSL #25              ;
  STR   R2,[R3,#28]                ; Set D/C High
  MCR   cp15,0,R3,c7,c10,5         ; Data Memory Buffer
  :
  LDR   R3,spiv
  :
  .isdone
  MOV   R0,R1
  ADD   R0,R0,#64
  CMP   R0,R5
  BHI   sendrem
  :
  ;SWI   256+67
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]
  :
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]
  :
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]
  :
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]
  :
  CMP   R1,R5
  BEQ   endsend
  :
  .bulk
  ;SWI   256+72
  MCR   cp15,0,R3,c7,c10,5
  .busyloop
  LDR   R2,[R3,#0]
  TST   R2,#1<<16
  BNE   isdone
  TST   R2,#1<<19
  ;MCREQ cp15,0,R3,c7,c0,4 ;wait for interrupt
  BEQ   busyloop
  :
  ;SWI   256+73
  MOV   R0,R1
  ADD   R0,R0,#48
  CMP   R0,R5
  BHI   sendrem
  :
  ;SWI   256+74
  LDR   R2,[R3,#4]
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]
  LDR   R2,[R3,#4]
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]
  LDR   R2,[R3,#4]
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]
  LDR   R2,[R3,#4]
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]
  :
  LDR   R2,[R3,#4]
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]
  LDR   R2,[R3,#4]
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]

  LDR   R0,[R1],#4

  STR   R0,[R3,#4]
  :
  LDR   R2,[R3,#4]
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]
  LDR   R2,[R3,#4]
  LDR   R0,[R1],#4

  CMP   R1,R5
  BNE   bulk
  :
  .endsend
  ;SWI   256+69
  .endloop
  MCR   CP15,0,R3,c7,c10,5
  LDR   R2,[R3,#0]
  TST   R2,#1<<17
  LDRNE R0,[R3,#4]
  TST   R2,#1<<16
  BEQ   endloop
  :
  ORR   R2,R2,#&30
  BIC   R2,R2,#&80
  STR   R2,[R3,#0]
  :
  SWI   "OS_LeaveOS"
  :
  MOV   PC,R14
  :

  ;SWI   256+68
  CMP   R1,R5
  BEQ   endsend
  LDR   R0,[R1],#4
  STR   R0,[R3,#4]
  BAL   sendrem
  :
  .gpiov
  EQUD  gpio%
  .spiv
  EQUD  spi%
  .SPI_AND_MASK
  EQUD  &0000000c
  .SPI_OR_MASK
  EQUD  &000006b0
  :

  .lineblt
  LDR   R1,[R0],#4
  MOV   R4,R1,LSR #8
  AND   R3,R1,#&ff
  AND   R4,R4,#&ff
  STR   R4,[R2],#4

  MOV   R4,R1,LSR #24

  BAL   lineblt
  :
] : NEXT pass%

DEF PROCspi(cmd%,addr%,end%)
REM PRINT "PROCspi(&";~cmd%;",&";~addr%;",&";~end%;")"
A%=cmd%:B%=addr%:F%=end%

ENDPROC

DEF PROClcdUpdate
LOCAL y%

  C%=y%+&95880:F%=y%:A%=USR lineblt
NEXT y%
PROCspi(&2c,dmabuffer%,dmabuffer%+614400)
ENDPROC

DEF PROCtest2
LOCAL x%,y%,p%
p%=spriteptr%+44
FOR y%=0 TO 239
  FOR x%=0 TO 319
    IF (x% AND &4)=(y% AND &4) THEN
      p%?0 = &ff
      p%?1 = &ff
      ELSE
      p%?0 = &00
      p%?1 = &00
    ENDIF
    p%=p%+2
  NEXT x%
NEXT y%
PROClcdUpdate
ENDPROC

  COLOUR c%

ENDPROC

DEF PROCvduLcdSprite
IF vdurestore%(0) THEN
  ERR 6,"Cannot switch"
  ELSE
  SYS spriteOp%,60+512,spritearea%,spriteptr%,savearea% TO vdurestore%(0),vdurestore%(1),vdurestore%(2),vdurestore%(3)
ENDIF
ENDPROC

DEF PROCvduRestore
IF vdurestore%(0) THEN
  SYS spriteOp%,vdurestore%(0),vdurestore%(1),vdurestore%(2),vdurestore%(3)
  vdurestore%(0)=0
  ELSE
  ERR 6,"Cannot restore"
ENDIF
ENDPROC

DEF PROCinitGPIO
LOCAL pass%
FOR pass%=0 TO 2 STEP 2
  P%=code% : [ OPT pass%
  .start
  SWI   "OS_EnterOS"
  LDR   R0, GPSEL0_OR_MASK
  LDR   R2,[R1,#0]
  ORR   R2,R2,R0
  LDR   R0, GPSEL0_AND_MASK
  AND   R2,R2,R0
  STR   R2,[R1,#0]
  LDR   R0,GPSEL1_OR_MASK
  LDR   R2,[R1,#4]
  ORR   R2,R2,R0
  LDR   R0,GPSEL1_AND_MASK
  AND   R2,R2,R0
  STR   R2,[R1,#4]
  ;LDR   R0,GPSEL2_OR_MASK
  ;LDR   R2,[R1,#8]
  ;ORR   R2,R2,R0
  ;LDR   R0,GPSEL2_AND_MASK
  ;AND   R2,R2,R0
  ;STR   R2,[R1,#8]

  MOV   R0,#%11<<6
  STR   R0,[R1,#&1c]

  MOV   R0,#%111<<9
  STR   R0,[R1,#&28]

  MOV   R0,#&3c
  STR   R0,[R3,#0]

  MOV   R0,#spiclk%
  STR   R0,[R3,#&8]

  SWI   "OS_LeaveOS"
  MOV   PC,R14
  :
  .GPSEL0_OR_MASK
  EQUD  &24800000
  :
  .GPSEL0_AND_MASK
  EQUD  &249fffff
  :
  .GPSEL1_OR_MASK
  EQUD  &00000024
  :
  .GPSEL1_AND_MASK
  EQUD  &2fffffe4
  :
  .GPSEL2_OR_MASK
  EQUD  &00024000
  :
  .GPSEL2_AND_MASK
  EQUD  &fffc0fff
] : NEXT pass%
B%=gpio%:D%=spi%
CALL start
SYS "GPIO_WriteMode",8,1
SYS "GPIO_WriteData",18,1
SYS "GPIO_WriteMode",25,1
SYS "GPIO_WriteMode",27,1
ENDPROC

DEF PROClcdSetup
LOCAL DATA
RESTORE +0
DATA &136,&00
DATA &13A,&05
DATA &131
DATA &12A,&00,&00,&01,&3F
DATA &12B,&00,&00,&00,&EF
DATA &1B2,&0C,&0C,&00,&33,&33
DATA &1B7,&35
DATA &1BB,&1F
DATA &1C0,&2C
DATA &1C2,&01
DATA &1C3,&12
DATA &1C4,&20
DATA &1C6,&2F
DATA &1D0,&A4,&A1
DATA &1E0,&D0,&08,&11,&08,&0C,&15,&39,&33,&50,&36,&13,&14,&29,&2D
DATA &1E1,&D0,&08,&10,&08,&06,&06,&39,&44,&51,&0B,&16,&14,&2F,&31
DATA &121
DATA &111
DATA -1
PROClcdSendInitData
PROCsleep(0.500)
PROCspi(&29,code%,code%)
RESTORE DATA
ENDPROC

DEF PROClcdSendInitData
LOCAL ptr%
ptr%=code%
READ !ptr%
WHILE -1<!ptr%
  !ptr%=&ff AND !ptr%
  ptr%=ptr%+4
  READ !ptr%
  WHILE (&100 AND !ptr%)=0
    ptr%=ptr%+4
    READ !ptr%
  ENDWHILE
  PROCspi(!code%,code%+4,ptr%)
  !code%=!ptr%
  ptr%=code%
ENDWHILE
ENDPROC

DEF PROCsleep(d)
LOCAL t%
t%=d*100+TIME:REPEAT:UNTIL TIME>=t%
ENDPROC

DEF PROClcdReset
  SYS "GPIO_WriteData",18,0
  SYS "GPIO_WriteData",8,1
  PROCsleep(0.010)
  SYS "GPIO_WriteData",27,1
  PROCsleep(0.005)
  SYS "GPIO_WriteData",27,0
  PROCsleep(0.020)
  SYS "GPIO_WriteData",27,1
  PROCsleep(0.150)
  SYS "GPIO_WriteData",8,0
  SYS "GPIO_WriteData",18,1
ENDPROC

DEF PROCsetWindow(X0%,Y0%,X1%,Y1%)
code%!0  = (X0% >> 8) AND &FF
code%!4  = (X0%)      AND &FF
code%!8  = (X1% >> 8) AND &FF
code%!12 = (X1%)      AND &FF
PROCspi(&2B,code%,code%+16)

code%!0  = (Y0% >> 8) AND &FF
code%!4  = (Y0%)      AND &FF
code%!8  = (Y1% >> 8) AND &FF
code%!12 = (Y1%)      AND &FF
PROCspi(&2C,code%,code%+16)
ENDPROC
